TabDisplay ​v0.0.0

TabDisplay formats Java player-list names, headers, and footers. It leaves Bedrock lists unchanged.

See the plugin guide and API reference on the documentation site.

Installation ​

Download tab-display.wasm from the plugin's GitHub release, put it in your server's plugins/ folder, and restart Pumpkin. On the first load, a full restart lets Pumpkin ask for the plugin's permissions; hot reload cannot prompt for them.

PumpkinPIE placeholders are optional and come from a separate external plugin. TabDisplay does not build or bundle PumpkinPIE; install a compatible provider from the PumpkinPIE releases.

Permissions ​

Pumpkin asks you to approve these on the server console when it loads the plugin. It skips the question when the [plugins] table in pumpkin.toml lists all of them in allowed_permissions or sets ask_permission_confirmation = false. Pumpkin remembers your answer and asks again after the plugin file or this list changes.

PermissionWhy
fs.read.dataRead TabDisplay settings from the plugin data folder.
fs.write.dataCreate and update TabDisplay settings in the plugin data folder.
http.outboundCheck Pumpkin Market for plugin updates.

A fresh install gets this file:
toml
# TabDisplay configuration.
# This file is managed by the plugin: when it starts, settings that were added are written here with
# their defaults and settings that were removed are dropped. Your values are kept; comments you add
# are not. Changes apply after `/tabdisplay reload` or a server restart.

# Plugin refresh and workload limits.
[general]

# Format version. Do not change this value.
version = 1

# Apply configured Java tab display changes.
enabled = true

# Ticks between full display refreshes.
refresh_ticks = 20

# Maximum player snapshots processed per refresh.
max_players = 500

# PumpkinPIE placeholder handling.
[placeholders]

# Pumpkin plugin that supplies placeholder values.
provider = "PumpkinPIE"

# Text to show when PumpkinPIE is unavailable: `literal` keeps tokens, `empty` removes them.
missing_provider = "literal"

# Text to show for unresolved identifiers: `literal` keeps tokens, `empty` removes them.
unresolved = "literal"

# Personalized header lines. Each player sees their own placeholder context.
[header]

# Header lines in display order. Use a TOML array of inline tables, for example [{ text = "Welcome
# %player_name%", color = "#FFB347" }].
lines = [{ text = "Welcome %player_name%", color = "#FFB347" }]

# Personalized footer lines. Each player sees their own placeholder context.
[footer]

# Footer lines in display order. Use a TOML array of inline tables, for example [{ text =
# "%server_online% players online", color = "#AAAAAA" }].
lines = [{ text = "%server_online% players online", color = "#AAAAAA" }]

# Global real-player row formatting. Placeholder context is the listed subject.
[entry]

# Apply a custom name to real player rows.
enabled = true

# Name template for each listed player.
text = "%player_name%"

# Optional RGB text color in `#RRGGBB` form.
color = "#FFFFFF"

# Render the row name in bold.
bold = false

# Render the row name in italics.
italic = false

# Underline the row name.
underlined = false

# Strike through the row name.
strikethrough = false

# Global Java player-list ordering. Rules stay disabled until native client order is verified.
[sorting]

# Reserved sorting keys. Non-empty rules are rejected until native Java order direction is verified.
rules = []

# Java player-list numeric objective. It is not enabled until client behavior is verified.
[objective]

# Show a numeric objective in the Java player list.
enabled = false

# Value source reserved for the verified objective adapter.
value = "ping"

# Native Bedrock features.
[bedrock]

# Show a numeric objective in the native Bedrock player list.
objective_enabled = false
